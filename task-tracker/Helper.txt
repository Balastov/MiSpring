Краткая памятка для управления

# 1. Перезапуск приложения
sudo systemctl restart task-tracker

# 2. Перезапуск Nginx
sudo systemctl restart nginx 

# 3. Проверка статуса всех сервисов
sudo systemctl status task-tracker nginx certbot.timer

# 4. Просмотр логов
sudo journalctl -u task-tracker -f          # Логи приложения
sudo tail -f /var/log/nginx/error.log      # Логи ошибок Nginx
sudo certbot renew --dry-run               # Тест обновления SSL

# 5. Обновление кода (деплой)
cd ~/projects/MiSpring
git pull origin main
sudo systemctl restart task-tracker

# 5.1. Если были изменения в базе, но данные на сервере нужно сохранить
cd ~/projects/MiSpring && git checkout -- task-tracker/tasks.db && git pull origin main && sudo systemctl restart task-tracker
Если вам важно сохранить данные в БД на сервере, тогда лучше так:
cd ~/projects/MiSpring && git stash push task-tracker/tasks.db && git pull origin main && git stash pop && sudo systemctl restart task-tracker



# 6. Перенос БД в /var/lib (одноразово при первом деплое)
sudo mkdir -p /var/lib/task-tracker
sudo mv ~/projects/MiSpring/task-tracker/tasks.db /var/lib/task-tracker/tasks.db
sudo chown user1:user1 /var/lib/task-tracker/tasks.db
